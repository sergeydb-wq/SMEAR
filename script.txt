UserInputService.InputBegan:Connect(function(input, processed)

    --------------------------------------------------
    -- SETTING NEW KEY
    --------------------------------------------------

    if waitingForBind then

        if input.KeyCode == Enum.KeyCode.Escape then

            waitingForBind = nil

            updateKeybindGUI()

            return
        end

        if input.KeyCode ~= Enum.KeyCode.Unknown then

            config.binds[waitingForBind] =
                input.KeyCode

            waitingForBind = nil

            updateKeybindGUI()
            updateMainGUI()

            return
        end
    end

    --------------------------------------------------
    -- NORMAL BINDS
    --------------------------------------------------

    if processed then
        return
    end

    if input.KeyCode == config.binds.Combat then

        config.combatEnabled =
            not config.combatEnabled

        updateMainGUI()

    elseif input.KeyCode == config.binds.ESP then

        config.espEnabled =
            not config.espEnabled

        updateMainGUI()

    elseif input.KeyCode == config.binds.Friends then

        if FriendFrame then

            FriendFrame.Visible =
                not FriendFrame.Visible
        end
    end
end)

local ESPFolder = Instance.new("Folder")
ESPFolder.Name = "ESP"
ESPFolder.Parent = ScreenGui

local ESPObjects = {}

local function removeESP(player)

    local data =
        ESPObjects[player]

    if not data then
        return
    end

    if data.Highlight then
        pcall(function()
            data.Highlight:Destroy()
        end)
    end

    if data.Billboard then
        pcall(function()
            data.Billboard:Destroy()
        end)
    end

    if data.HealthConnection then
        pcall(function()
            data.HealthConnection:Disconnect()
        end)
    end

    ESPObjects[player] = nil
end

local function createESP(player)

    if player == LocalPlayer then
        return
    end

    removeESP(player)

    if not config.espEnabled then
        return
    end

    local character =
        player.Character

    if not character then
        return
    end

    local humanoid =
        character:FindFirstChildOfClass("Humanoid")

    local root =
        character:FindFirstChild(
            "HumanoidRootPart"
        )

    if not humanoid or not root then
        return
    end

    --------------------------------------------------
    -- HIGHLIGHT
    --------------------------------------------------

    local highlight =
        Instance.new("Highlight")

    highlight.Name = "PlayerESP"

    highlight.Adornee =
        character

    highlight.DepthMode =
        Enum.HighlightDepthMode.AlwaysOnTop

    highlight.FillTransparency = 0.55
    highlight.OutlineTransparency = 0

    if isFriend(player) then

        highlight.FillColor =
            Color3.fromRGB(
                50,
                220,
                80
            )

        highlight.OutlineColor =
            Color3.fromRGB(
                120,
                255,
                130
            )

    else

        highlight.FillColor =
            Color3.fromRGB(
                255,
                50,
                50
            )

        highlight.OutlineColor =
            Color3.fromRGB(
                255,
                120,
                120
            )
    end

    highlight.Parent = ESPFolder

    --------------------------------------------------
    -- HP
    --------------------------------------------------

    local Billboard =
        Instance.new("BillboardGui")

    Billboard.Name = "HealthESP"

    Billboard.Adornee =
        root

    Billboard.Size =
        UDim2.new(0, 180, 0, 40)

    Billboard.StudsOffset =
        Vector3.new(0, 3.5, 0)

    Billboard.AlwaysOnTop = true
    Billboard.MaxDistance = 1000

    Billboard.Parent = ESPFolder

    local HPText =
        Instance.new("TextLabel")

    HPText.Size =
        UDim2.new(1, 0, 1, 0)

    HPText.BackgroundTransparency = 1

    HPText.Font =
        Enum.Font.GothamBold

    HPText.TextSize = 15

    HPText.TextStrokeTransparency = 0

    HPText.Parent = Billboard

    local function updateHP()

        if not humanoid.Parent then
            return
        end

        local hp =
            math.max(
                0,
                math.floor(humanoid.Health)
            )

        local maxHP =
            math.max(
                1,
                math.floor(humanoid.MaxHealth)
            )

        HPText.Text =
            string.format(
                "%s | HP: %d/%d",
                player.DisplayName,
                hp,
                maxHP
            )

        if isFriend(player) then

            HPText.TextColor3 =
                Color3.fromRGB(
                    80,
                    255,
                    100
                )

        else

            HPText.TextColor3 =
                Color3.fromRGB(
                    255,
                    80,
                    80
                )
        end
    end

    updateHP()

    local healthConnection =
        humanoid.HealthChanged:Connect(
            updateHP
        )

    ESPObjects[player] = {
        Highlight = highlight,
        Billboard = Billboard,
        HealthConnection = healthConnection
    }
end

-- TARGETS
--------------------------------------------------

local function getTargets()

    local targets = {}

    local char =
        LocalPlayer.Character

    local root =
        char and char:FindFirstChild(
            "HumanoidRootPart"
        )

    if not root then
        return targets
    end

    for _, obj in ipairs(
        Workspace:GetChildren()
    ) do

        if obj:IsA("Model")
            and obj ~= char then

            local humanoid =
                obj:FindFirstChild("Humanoid")

            local hitbox =
                obj:FindFirstChild(
                    "PlayerHitbox"
                )

            if humanoid
                and hitbox
                and humanoid.Health > 0 then

                local distance =
                    (
                        root.Position -
                        hitbox.Position
                    ).Magnitude

                if distance <= config.range then

                    local player =
                        Players:GetPlayerFromCharacter(
                            obj
                        )

                    -- FRIENDS ARE IGNORED
                    if not player
                        or not Friends[player.UserId] then

                        table.insert(
                            targets,
                            {
                                Model = obj,
                                Player = player,
                                Humanoid = humanoid,
                                Hitbox = hitbox,
                                Distance = distance
                            }
                        )
                    end
                end
            end
        end
    end

    table.sort(
        targets,
        function(a, b)
            return a.Distance <
                b.Distance
        end
    )

    return targets
end

--------------------------------------------------
-- ATTACK COOLDOWN
--------------------------------------------------

local lastAttack = {}

local function canAttack(target)

    local now =
        os.clock()

    local previous =
        lastAttack[target]

    if not previous
        or now - previous >=
            config.targetCooldown then

        lastAttack[target] = now

        return true
    end

    return false
end

--------------------------------------------------

local attemptId = 0

local function attackTarget(
    target,
    weaponName,
    weaponType
)

    if not target then
        return
    end

    if target.Player
        and Friends[target.Player.UserId] then
        return
    end

    if not target.Humanoid
        or not target.Hitbox then
        return
    end

    if target.Humanoid.Health <= 0 then
        return
    end

    if not canAttack(target.Model) then
        return
    end

    local char =
        LocalPlayer.Character

    if not char then
        return
    end

    local root =
        char:FindFirstChild(
            "HumanoidRootPart"
        )

    if not root then
        return
    end

    pcall(function()

        if weaponType == "Mace" then

            attemptId =
                (attemptId + 1) % 10000

            local originPos =
                root.Position +
                Vector3.new(0, 15, 0)

            local targetPos =
                target.Hitbox.Position

            local difference =
                targetPos - originPos

            if difference.Magnitude <= 0 then
                return
            end

            local aimDir =
                difference.Unit

            CombatEvent:FireServer(
                target.Humanoid,
                target.Hitbox,
                1,
                "Mace",
                {
                    SmashMissTipPosition =
                        targetPos,

                    SmashMissRayOrigin =
                        originPos,

                    FallSpeed = 600,

                    AimDirection =
                        aimDir,

                    ClientMaceSmashHitPosition =
                        targetPos,

                    SourceCharacter =
                        char,

                    FallDistance = 800,

                    SmashAttemptId =
                        attemptId,

                    ClientMaceSmashHurtbox =
                        true
                }
            )

        elseif weaponType == "Standard" then

            CombatEvent:FireServer(
                target.Humanoid,
                target.Hitbox,
                1,
                weaponName,
                {
                    SourceCharacter =
                        char,

                    IsFalling = true
                }
            )
        end
    end)
end

local function makeDraggable(frame)

    local dragging = false
    local dragStart
    local startPos

    frame.InputBegan:Connect(function(input)

        if input.UserInputType ==
            Enum.UserInputType.MouseButton1 then

            dragging = true

            dragStart =
                input.Position

            startPos =
                frame.Position

            input.Changed:Connect(
                function()

                    if input.UserInputState ==
                        Enum.UserInputState.End then

                        dragging = false
                    end
                end
            )
        end
    end)

    UserInputService.InputChanged:Connect(
        function(input)

            if not dragging then
                return
            end

            if input.UserInputType ==
                Enum.UserInputType.MouseMovement then

                local delta =
                    input.Position -
                    dragStart

                frame.Position =
                    UDim2.new(
                        startPos.X.Scale,
                        startPos.X.Offset +
                            delta.X,

                        startPos.Y.Scale,
                        startPos.Y.Offset +
                            delta.Y
                    )
            end
        end
    )
end

makeDraggable(Main)
makeDraggable(FriendFrame)
makeDraggable(KeybindFrame)
